const {chromium} = require('playwright');
const assert = require('node:assert/strict');
const fs = require('fs');
const path = require('path');
(async () => {
 const browser = await chromium.launch({executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe',headless:true});
 try {
 const page = await browser.newPage();
 const errors=[];page.on('pageerror',e=>errors.push(e.message));
 await page.route('https://cdn.jsdelivr.net/npm/mermaid@11.12.0/dist/**', async route => {
  const name=route.request().url().split('/dist/')[1];
  await route.fulfill({body:fs.readFileSync(path.join(__dirname,'node_modules/mermaid/dist',name)),contentType:'application/javascript'});
 });
 await page.route(/https:\/\/(cdn.jsdelivr.net|cdnjs.cloudflare.com)\/.*bootstrap.*\.(js|css)/, async route=>{
  const kind=route.request().url().endsWith('.js')?'js/bootstrap.bundle.min.js':'css/bootstrap.min.css';
  await route.fulfill({body:fs.readFileSync(path.join(__dirname,'node_modules/bootstrap/dist',kind)),contentType:kind.startsWith('js')?'application/javascript':'text/css'});
 });
 await page.goto('http://127.0.0.1:55433/login/');
 page.on('dialog',dialog=>dialog.accept());
 await page.fill('#username','browser_fixture');await page.fill('#password','browser-fixture-password');
 await page.click('button[type=submit]');await page.waitForURL('http://127.0.0.1:55433/');
 await page.waitForFunction(()=>window.mermaid && document.querySelector('#generatorCategory').options.length===4);
 assert.equal(await page.evaluate(()=>window.injected),undefined);
 assert.equal(await page.locator('#chatsContainer img').count(),0);
 for(const mode of ['mapa-mental','fluxograma']){
  await page.selectOption('#generatorCategory',{label:'Navegador'});
  await page.selectOption('#generatorMode',mode);
  const response=page.waitForResponse(r=>r.url()==='http://127.0.0.1:55433/'&&r.request().method()==='POST');
  await page.click('#generateBtn');const result=await (await response).json();
  assert.equal(result.modalidade,mode);assert.equal(result.total_chats,1);
  await page.waitForSelector('#modalMermaidContent svg');
  await page.waitForSelector('#modalMermaidContent a');
  const evidence=await (await page.request.get(`http://127.0.0.1:55433/output/${result.output_id}/export/json/`)).json();
  assert.equal(evidence.mensagens[0].texto.includes('mais'),false);
  assert.ok(evidence.mensagens[0].texto.includes('Chaves primárias'));
  assert.equal(evidence.modalidade,mode);
  const downloadPromise = page.waitForEvent('download', {timeout: 10000});
  await page.getByRole('button',{name:'Exportar PNG'}).click();
  const download = await downloadPromise.catch(async error => {
    console.error('Exportação:', await page.locator('#errorToast').textContent());
    throw error;
  });
  const imagePath = path.join(__dirname,`${mode}.png`);
  await download.saveAs(imagePath);
  assert.ok(fs.readFileSync(imagePath).subarray(0,8).equals(Buffer.from([137,80,78,71,13,10,26,10])));
  const mmd = await page.request.get(`http://127.0.0.1:55433/output/${result.output_id}/export/mmd/`);
  assert.equal(await mmd.text(),evidence.mermaid);
  await page.waitForFunction(()=>!document.querySelector('#generateBtn').disabled);
  await page.locator('#generatedContentModal button[data-bs-dismiss=modal]').last().click();
  await page.waitForSelector('#generatedContentModal',{state:'detached'});
  console.log('PASS fluxo completo, renderer SVG, export JSON e botão restaurado:',mode);
 }
 await page.selectOption('#generatorCategory',{label:'Invalida'});await page.selectOption('#generatorMode','fluxograma');
 await page.click('#generateBtn');
 await page.waitForSelector('#modalMermaidContent pre');
 assert.ok((await page.locator('#modalMermaidContent').textContent()).includes('Falha de renderização'));
 assert.ok((await page.locator('#modalMermaidContent pre').textContent()).includes('A[sem fechamento'));
 await page.locator('#generatedContentModal button[data-bs-dismiss=modal]').last().click();
 await page.waitForSelector('#generatedContentModal',{state:'detached'});
 console.log('PASS sintaxe inválida: falha de renderização diferenciada e texto preservado');
 await page.selectOption('#generatorCategory',{label:'Vazia'});
 const failure=page.waitForResponse(r=>r.url()==='http://127.0.0.1:55433/'&&r.request().method()==='POST');
 await page.click('#generateBtn');assert.equal((await failure).status(),502);
 await page.waitForFunction(()=>!document.querySelector('#generateBtn').disabled);
 assert.equal(await page.locator('#generatedContentModal').count(),0);
 assert.ok((await page.locator('#errorToast').textContent()).includes('conteúdo vazio'));
 console.log('PASS inferência vazia: HTTP 502, mensagem útil, sem modal, botão restaurado');
 assert.deepEqual(errors,[]);
 console.log('PASS texto HTML das mensagens não executado; sem exceções JavaScript não tratadas');
 await page.screenshot({path:path.join(__dirname,'frontend.png')});
 } finally {await browser.close();}
})().catch(e=>{console.error(e);process.exitCode=1;});
